cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{DEVKITPRO}/cmake/Switch.cmake")
endif()
project(papa_walkman VERSION 0.2.0 LANGUAGES C CXX)
add_executable(papa-walkman source/main.cpp source/core.cpp source/decoder.cpp source/engine.cpp source/audio.cpp source/ui.cpp)
target_compile_features(papa-walkman PRIVATE cxx_std_17)
target_compile_options(papa-walkman PRIVATE -Wall -Wextra -O2 -fno-exceptions -fno-rtti)
target_include_directories(papa-walkman PRIVATE source)
target_include_directories(papa-walkman SYSTEM PRIVATE third_party/minimp3 third_party/stb)
nx_generate_nacp(OUTPUT papa-walkman.nacp NAME "爬爬的随身听" AUTHOR "Papa" VERSION "0.2.0")
nx_create_nro(papa-walkman ICON "${CMAKE_CURRENT_SOURCE_DIR}/assets/icon.jpg" NACP papa-walkman.nacp)
