cmake_minimum_required(VERSION 3.20)
option(PAPA_HOST_TESTS "Build portable host verification" OFF)
if(NOT PAPA_HOST_TESTS AND NOT DEFINED CMAKE_TOOLCHAIN_FILE)
    set(CMAKE_TOOLCHAIN_FILE "$ENV{DEVKITPRO}/cmake/Switch.cmake")
endif()
project(papa_walkman VERSION 0.3.0 LANGUAGES C CXX)
if(PAPA_HOST_TESTS)
    foreach(flags CMAKE_C_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELEASE CMAKE_C_FLAGS_RELWITHDEBINFO CMAKE_CXX_FLAGS_RELWITHDEBINFO)
        string(REGEX REPLACE "[/-]DNDEBUG" "" ${flags} "${${flags}}")
    endforeach()
endif()
add_library(papa_portable STATIC source/core.cpp source/media.cpp source/stream.cpp source/decoder.cpp source/engine.cpp source/app_model.cpp source/assets.cpp source/runtime.cpp source/pause_gate.cpp third_party/yyjson/yyjson.c)
target_compile_features(papa_portable PUBLIC cxx_std_17)
target_include_directories(papa_portable PUBLIC source)
target_include_directories(papa_portable SYSTEM PUBLIC third_party/minimp3 third_party/yyjson)
target_include_directories(papa_portable SYSTEM PRIVATE third_party/stb)
if(PAPA_HOST_TESTS)
    if(NOT PAPA_CURL_INCLUDE OR NOT PAPA_CURL_LIBRARY)
        message(FATAL_ERROR "Host tests require PAPA_CURL_INCLUDE and PAPA_CURL_LIBRARY; use tools/test-host.ps1")
    endif()
    target_include_directories(papa_portable SYSTEM PRIVATE "${PAPA_CURL_INCLUDE}")
    target_link_libraries(papa_portable PUBLIC "${PAPA_CURL_LIBRARY}")
    target_compile_definitions(papa_portable PUBLIC _CRT_SECURE_NO_WARNINGS)
    target_compile_definitions(papa_portable PRIVATE NOMINMAX)
    if(MSVC)
        target_compile_options(papa_portable PUBLIC /utf-8)
    endif()
    enable_testing()
    foreach(name core path decoder engine media app_model runtime pause_gate)
        add_executable(${name}_tests tests/${name}_tests.cpp)
        target_link_libraries(${name}_tests PRIVATE papa_portable)
        if(name STREQUAL "core")
            add_test(NAME ${name} COMMAND ${name}_tests build-host/fixtures)
        elseif(name STREQUAL "decoder" OR name STREQUAL "engine")
            add_test(NAME ${name} COMMAND ${name}_tests tests/tone.mp3)
        else()
            add_test(NAME ${name} COMMAND ${name}_tests)
        endif()
        set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    endforeach()
    add_executable(ui_tests tests/ui_tests.cpp source/ui.cpp)
    target_include_directories(ui_tests SYSTEM PRIVATE third_party/stb)
    target_link_libraries(ui_tests PRIVATE papa_portable)
    add_test(NAME ui COMMAND ui_tests)
    set_tests_properties(ui PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(papa_preview tools/preview.cpp source/ui.cpp)
    target_include_directories(papa_preview SYSTEM PRIVATE third_party/stb)
    target_link_libraries(papa_preview PRIVATE papa_portable)
    add_executable(stream_tests tests/stream_tests.cpp)
    target_link_libraries(stream_tests PRIVATE papa_portable)
    add_executable(assets_tests tests/assets_tests.cpp)
    target_include_directories(assets_tests SYSTEM PRIVATE third_party/stb)
    target_link_libraries(assets_tests PRIVATE papa_portable)
    add_test(NAME assets COMMAND assets_tests)
    set_tests_properties(assets PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
    add_executable(audio_tests tests/audio_tests.cpp source/audio.cpp)
    target_link_libraries(audio_tests PRIVATE papa_portable)
    add_test(NAME audio COMMAND audio_tests tests/tone.mp3)
    set_tests_properties(audio PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
else()
    target_compile_definitions(papa_portable PUBLIC CURL_STATICLIB)
    target_compile_options(papa_portable PRIVATE -Wall -Wextra -O2 "$<$<COMPILE_LANGUAGE:CXX>:-fno-exceptions>" "$<$<COMPILE_LANGUAGE:CXX>:-fno-rtti>")
    target_include_directories(papa_portable SYSTEM PRIVATE "$ENV{DEVKITPRO}/portlibs/switch/include")
    target_link_libraries(papa_portable PUBLIC curl z)
    add_executable(papa-walkman source/main.cpp source/audio.cpp source/ui.cpp)
    target_compile_options(papa-walkman PRIVATE -Wall -Wextra -O2 -fno-exceptions -fno-rtti)
    target_include_directories(papa-walkman SYSTEM PRIVATE third_party/stb)
    target_link_libraries(papa-walkman PRIVATE papa_portable)
    nx_generate_nacp(OUTPUT papa-walkman.nacp NAME "爬爬的随身听" AUTHOR "Papa" VERSION "0.3.0")
    nx_create_nro(papa-walkman ICON "${CMAKE_CURRENT_SOURCE_DIR}/assets/icon.jpg" NACP papa-walkman.nacp)
endif()
